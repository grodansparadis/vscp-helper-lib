cmake_minimum_required(VERSION 3.14)
project(vscp_helper_dependency_check NONE)

if(NOT DEFINED VSCP_HELPER_SOURCE_DIR)
  message(FATAL_ERROR "VSCP_HELPER_SOURCE_DIR is required")
endif()
if(EXISTS "${FETCHCONTENT_SOURCE_DIR_VSCP_MONGOOSE}")
  message(FATAL_ERROR "The unavailable Mongoose source path must not exist")
endif()

include("${VSCP_HELPER_SOURCE_DIR}/cmake/VscpHelperDependencies.cmake")
file(SHA256 "${vscp_SOURCE_DIR}/src/vscp/common/vscpremotetcpif.h" original_client_header_hash)
file(SHA256 "${vscp_SOURCE_DIR}/src/vscp/common/vscpremotetcpif.cpp" original_client_hash)
file(SHA256 "${vscp_SOURCE_DIR}/src/vscp/common/vscphelper.cpp" original_helper_hash)
set(_logging_source "${vscp_SOURCE_DIR}/src/vscp/common/vscphelperlogging.cpp")
set(_logging_header "${vscp_SOURCE_DIR}/src/vscp/common/vscphelperlogging.h")
if(EXISTS "${_logging_source}" AND EXISTS "${_logging_header}")
  file(SHA256 "${_logging_source}" original_logging_source_hash)
  file(SHA256 "${_logging_header}" original_logging_header_hash)
  if(EXISTS "${vscp_SOURCE_DIR}/src/vscp/common/vscphelperlogging-forward.cpp")
    file(SHA256 "${vscp_SOURCE_DIR}/src/vscp/common/vscphelperlogging-forward.cpp"
         original_logging_forward_hash)
    file(SHA256 "${vscp_SOURCE_DIR}/src/vscp/common/vscphelperlogging-internal.h"
         original_logging_internal_hash)
  endif()
endif()
include("${VSCP_HELPER_SOURCE_DIR}/cmake/VscpHelperTransport.cmake")
file(SHA256 "${vscp_SOURCE_DIR}/src/vscp/common/vscpremotetcpif.h" current_client_header_hash)
file(SHA256 "${vscp_SOURCE_DIR}/src/vscp/common/vscpremotetcpif.cpp" current_client_hash)
file(SHA256 "${vscp_SOURCE_DIR}/src/vscp/common/vscphelper.cpp" current_helper_hash)
if(NOT original_client_hash STREQUAL current_client_hash OR
   NOT original_client_header_hash STREQUAL current_client_header_hash)
  message(FATAL_ERROR "The authenticated-framing patch must not modify fetched sources")
endif()
if(NOT original_helper_hash STREQUAL current_helper_hash)
  message(FATAL_ERROR "The logging patch must not modify fetched helper sources")
endif()
if(DEFINED original_logging_source_hash)
  file(SHA256 "${_logging_source}" current_logging_source_hash)
  file(SHA256 "${_logging_header}" current_logging_header_hash)
  file(SHA256 "${VSCP_HELPER_LOGGING_DIR}/vscphelperlogging.cpp" copied_logging_source_hash)
  file(SHA256 "${VSCP_HELPER_LOGGING_DIR}/vscphelperlogging.h" copied_logging_header_hash)
  if(NOT original_logging_source_hash STREQUAL current_logging_source_hash OR
     NOT original_logging_header_hash STREQUAL current_logging_header_hash OR
     NOT original_logging_source_hash STREQUAL copied_logging_source_hash OR
     NOT original_logging_header_hash STREQUAL copied_logging_header_hash)
    message(FATAL_ERROR "The upstream logger must be used without modifying its sources")
  endif()
endif()
if(DEFINED original_logging_forward_hash)
  file(SHA256 "${vscp_SOURCE_DIR}/src/vscp/common/vscphelperlogging-forward.cpp"
       current_logging_forward_hash)
  file(SHA256 "${vscp_SOURCE_DIR}/src/vscp/common/vscphelperlogging-internal.h"
       current_logging_internal_hash)
  if(NOT original_logging_forward_hash STREQUAL current_logging_forward_hash OR
     NOT original_logging_internal_hash STREQUAL current_logging_internal_hash)
    message(FATAL_ERROR "The private logger patch must not modify fetched sources")
  endif()
  file(READ "${VSCP_HELPER_CLIENT_DIR}/vscphelperlogging-internal.h" private_logging_header)
  string(FIND "${private_logging_header}" "#if defined(VSCP_HELPER_LOGGING_PRIVATE)"
         private_logging_position)
  if(private_logging_position EQUAL -1 OR
     EXISTS "${VSCP_HELPER_CLIENT_DIR}/vscphelperlogging.h")
    message(FATAL_ERROR "The private logger must use a private backend and the canonical public header")
  endif()
endif()
file(READ "${VSCP_HELPER_CLIENT_DIR}/vscpremotetcpif.h" patched_client_header)
string(FIND "${patched_client_header}" "setBinarySecurity" security_method_position)
if(security_method_position EQUAL -1)
  message(FATAL_ERROR "The build-local client does not include authenticated framing")
endif()
file(READ "${VSCP_HELPER_CLIENT_DIR}/vscpremotetcpif.cpp" patched_client_source)
foreach(_security_code "VscpRemoteTcpIf::setBinarySecurity"
                       "vscp_encryptFrameV1" "vscp_decryptFrameV1"
                       "OPENSSL_cleanse(m_binaryKey.data(), m_binaryKey.size())"
                       "OPENSSL_cleanse(m_binaryHmacKey.data(), m_binaryHmacKey.size())")
  string(FIND "${patched_client_source}" "${_security_code}" security_code_position)
  if(security_code_position EQUAL -1)
    message(FATAL_ERROR "The build-local client is missing authenticated framing code: ${_security_code}")
  endif()
endforeach()
string(FIND "${patched_client_source}" "vscphelper_logTransportError(format, arguments)"
       logging_method_position)
if(logging_method_position EQUAL -1)
  message(FATAL_ERROR "The build-local client does not route transport diagnostics")
endif()
file(READ "${VSCP_HELPER_CLIENT_DIR}/vscphelper.cpp" patched_helper_source)
string(FIND "${patched_helper_source}" "VSCPHLP_LOG_ERROR" helper_logging_position)
string(FIND "${patched_helper_source}" "vscphelper_logFunctionFailure" helper_failure_position)
if(helper_logging_position EQUAL -1 AND helper_failure_position EQUAL -1)
  message(FATAL_ERROR "The build-local helper does not route helper diagnostics")
endif()
file(READ "${VSCP_HELPER_LOGGING_DIR}/vscphelperlogging.h" patched_logging_header)
string(FIND "${patched_logging_header}" "vscphelper_configureLogging" logging_api_position)
if(logging_api_position EQUAL -1 OR NOT EXISTS "${VSCP_HELPER_LOGGING_DIR}/vscphelperlogging.cpp")
  message(FATAL_ERROR "The build-local VSCP logger is missing its implementation or API")
endif()

set(_crlf_source_root "${CMAKE_CURRENT_BINARY_DIR}/crlf-vscp")
set(_crlf_common "${_crlf_source_root}/src/vscp/common")
file(MAKE_DIRECTORY "${_crlf_common}")
set(_patched_files vscpremotetcpif.h vscpremotetcpif.cpp vscphelper.cpp)
foreach(_file IN LISTS _patched_files)
  string(MAKE_C_IDENTIFIER "${_file}" _key)
  file(SHA256 "${VSCP_HELPER_CLIENT_DIR}/${_file}" "_expected_${_key}")
  file(READ "${vscp_SOURCE_DIR}/src/vscp/common/${_file}" _contents)
  string(REPLACE "\r\n" "\n" _contents "${_contents}")
  string(REPLACE "\n" "\r\n" _contents "${_contents}")
  file(WRITE "${_crlf_common}/${_file}" "${_contents}")
  file(SHA256 "${_crlf_common}/${_file}" "_crlf_${_key}")
endforeach()
if(DEFINED original_logging_source_hash)
  set(_logging_files vscphelperlogging.cpp vscphelperlogging.h)
  if(EXISTS "${vscp_SOURCE_DIR}/src/vscp/common/vscphelperlogging-forward.cpp")
    list(APPEND _logging_files vscphelperlogging-forward.cpp vscphelperlogging-internal.h)
  endif()
  foreach(_file IN LISTS _logging_files)
    file(READ "${vscp_SOURCE_DIR}/src/vscp/common/${_file}" _contents)
    string(REPLACE "\r\n" "\n" _contents "${_contents}")
    string(REPLACE "\n" "\r\n" _contents "${_contents}")
    file(WRITE "${_crlf_common}/${_file}" "${_contents}")
  endforeach()
else()
  file(REMOVE "${_crlf_common}/vscphelperlogging.cpp" "${_crlf_common}/vscphelperlogging.h")
endif()

set(_original_vscp_source_dir "${vscp_SOURCE_DIR}")
set(vscp_SOURCE_DIR "${_crlf_source_root}")
include("${VSCP_HELPER_SOURCE_DIR}/cmake/VscpHelperTransport.cmake")
set(vscp_SOURCE_DIR "${_original_vscp_source_dir}")
foreach(_file IN LISTS _patched_files)
  string(MAKE_C_IDENTIFIER "${_file}" _key)
  file(SHA256 "${_crlf_common}/${_file}" _current_crlf_hash)
  file(SHA256 "${VSCP_HELPER_CLIENT_DIR}/${_file}" _current_patched_hash)
  if(NOT "${_crlf_${_key}}" STREQUAL "${_current_crlf_hash}")
    message(FATAL_ERROR "Patching a CRLF checkout must not modify ${_file}")
  endif()
  if(NOT "${_expected_${_key}}" STREQUAL "${_current_patched_hash}")
    message(FATAL_ERROR "CRLF checkout produced different patched contents for ${_file}")
  endif()
endforeach()

FetchContent_GetProperties(vscp_mongoose POPULATED mongoose_populated)
if(mongoose_populated)
  message(FATAL_ERROR "The helper library must not fetch Mongoose")
endif()
